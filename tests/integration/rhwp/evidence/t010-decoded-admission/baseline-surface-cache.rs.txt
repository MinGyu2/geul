//! Canvas 렌더러의 전역 이미지 캐시 수명 정책.
//!
//! `web_canvas.rs`의 `IMAGE_CACHE`(브라우저가 디코드하는 `HtmlImageElement`)와 `DECODED_CANVAS_CACHE`
//! (Rust가 디코드해 그린 detached `HtmlCanvasElement`)는 WASM 스레드 전역이고 문서와 수명이 다르다.
//! 같은 그림 바이트를 쓰는 여러 문서가 한 항목을 함께 쓰므로(내용 주소), 문서 하나를 닫는다고 항목을 지울
//! 근거가 없다. 그래서 문서별 해제 대신 전역 예산을 둔다.
//!
//! 비용(`bytes`)의 뜻: 디코드한 Canvas는 `너비 × 높이 × 4`(RGBA backing store **추정**), 브라우저 이미지는
//! data URL 길이(로드 전) 또는 data URL 길이 + 자연 크기의 RGBA 추정(로드 뒤), 실패를 기억하는 음성 항목은
//! [`NEGATIVE_ENTRY_BYTES`]다. 브라우저가 실제로 잡는 메모리(디코드 사본·GPU 사본·data URL 문자열)나 프로세스
//! RSS의 상한이 아니다 — 캐시가 붙잡아 GC가 놓지 못하게 하는 양의 추정이다.
//!
//! Resident accounting never exceeds bytes/entries. Pending admission can fail; the caller must
//! release a rejected value. Completion either becomes resident or transfers an explicit ready
//! lease to the caller. trim/budget reduction may cancel pending values; late callbacks are ignored.
//! Active DOM resources use reference-counted leases: Release must not invalidate another owner.

use std::collections::HashMap;

/// 디코드 실패를 기억하는 음성 항목의 계상 비용(바이트). 음성 항목도 항목 수 한도와 LRU를 따른다.
pub const NEGATIVE_ENTRY_BYTES: usize = 64;
/// 기본 예산: 디코드한 Canvas 합계 256 MiB(RGBA 약 67 Mpx), 브라우저 이미지 합계 64 MiB, 각 256개.
/// 실측 작업 집합(이미지가 많은 23쪽 문서 하나에 약 380 MB)보다 작다 — 입장 제한이 그 차이에서의 동작을 정한다.
pub const DEFAULT_DECODED_BYTES: usize = 256 << 20;
pub const DEFAULT_IMAGE_BYTES: usize = 64 << 20;
pub const DEFAULT_MAX_ENTRIES: usize = 256;
/// 요청 횟수 상한(포화). 작게 두어 오래 쓴 항목도 aging 몇 번이면 새 항목과 겨룰 수 있다.
const MAX_FREQUENCY: u8 = 15;

/// 그림 바이트의 내용 주소.
#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
pub struct ImageKey {
    fnv: u64,
    mix: u64,
    len: usize,
}

impl ImageKey {
    pub fn of(data: &[u8]) -> Self {
        // 전체 바이트의 FNV-1a 64(기존 키)에 길이와, 고르게 고른 바이트 64개의 두 번째 해시를 더한다.
        // 그리기마다 계산하므로 전체 바이트를 두 번 훑지 않는다.
        let mut fnv: u64 = 0xcbf2_9ce4_8422_2325;
        for &b in data {
            fnv ^= b as u64;
            fnv = fnv.wrapping_mul(0x0000_0100_0000_01b3);
        }
        let mut mix: u64 = 0x9e37_79b9_7f4a_7c15;
        if !data.is_empty() {
            for i in 0..64usize {
                let b = data[i * (data.len() - 1) / 63];
                mix = (mix ^ b as u64)
                    .wrapping_mul(0xff51_afd7_ed55_8ccd)
                    .rotate_left(29);
            }
        }
        Self {
            fnv,
            mix,
            len: data.len(),
        }
    }
}

/// 내보낸 값을 놓는 방법(Canvas: 크기 0, 이미지: 로드 알림 해제 후 src 비움). 시험에서는 기록만 한다.
pub trait Release {
    fn release(&self);
}

#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
pub struct CacheStats {
    pub entries: usize,
    pub negative_entries: usize,
    pub pending_entries: usize,
    pub transient_entries: usize,
    pub bytes: usize,
    pub max_bytes: usize,
    pub max_entries: usize,
    pub max_entry_bytes: usize,
    pub hits: u64,
    pub misses: u64,
    pub evictions: u64,
    /// 한 항목 한도를 넘어 보관하지 않았거나(디코드 Canvas) 일시 항목이 된(브라우저 이미지) 횟수.
    pub oversize: u64,
    /// 입장 제한으로 들이지 않은 횟수(그 값은 호출자가 그리고 놓았다).
    pub rejected: u64,
}

struct Entry<V> {
    value: Option<V>,
    bytes: usize,
    used: u64,
    pending: bool,
    transient: bool,
}

pub struct BoundedCache<V: Release> {
    entries: HashMap<ImageKey, Entry<V>>,
    bytes: usize,
    max_bytes: usize,
    max_entries: usize,
    max_entry_bytes: usize,
    tick: u64,
    stats: CacheStats,
    admission: bool,
    frequency: HashMap<ImageKey, u8>,
    since_aging: usize,
}

impl<V: Release + Clone> BoundedCache<V> {
    /// `max_entry_bytes`는 예산의 1/4: 큰 그림 하나가 다른 모든 항목을 밀어내지 않게 한다.
    /// `admission`이면 위의 입장 제한을 쓴다.
    pub fn new(max_bytes: usize, max_entries: usize, admission: bool) -> Self {
        Self {
            entries: HashMap::new(),
            bytes: 0,
            max_bytes,
            max_entries,
            max_entry_bytes: max_bytes / 4,
            tick: 0,
            stats: CacheStats::default(),
            admission,
            frequency: HashMap::new(),
            since_aging: 0,
        }
    }

    /// 있으면 `Some(값 또는 음성)`. 쓴 시각과 요청 횟수를 갱신한다. 로드 중인 값도 돌려준다(그릴 수 있는지는
    /// 호출자가 본다).
    pub fn get(&mut self, key: &ImageKey) -> Option<Option<V>> {
        self.tick += 1;
        self.count_request(key);
        match self.entries.get_mut(key) {
            Some(entry) => {
                entry.used = self.tick;
                self.stats.hits += 1;
                Some(entry.value.clone())
            }
            None => {
                self.stats.misses += 1;
                None
            }
        }
    }

    /// 넣는다. 보관했으면 true. 너무 크거나 입장 제한에 걸려 보관하지 않으면 false — 값은 호출자 것이다(그린 뒤
    /// 놓는다). 음성 항목은 늘 들인다(작고, 실패한 디코드를 되풀이하지 않게 한다).
    pub fn insert(&mut self, key: ImageKey, value: Option<V>, bytes: usize) -> bool {
        let negative = value.is_none();
        let bytes = if negative {
            NEGATIVE_ENTRY_BYTES
        } else {
            bytes
        };
        self.remove(&key);
        if bytes > self.max_bytes
            || self.max_entries == 0
            || (!negative && bytes > self.max_entry_bytes)
        {
            self.stats.oversize += 1;
            return false;
        }
        if !negative && self.admission && !self.admits(&key, bytes) {
            self.stats.rejected += 1;
            return false;
        }
        self.make_room(bytes, 1);
        if !self.fits(bytes, 1) {
            return false;
        }
        self.put(key, value, bytes, false);
        true
    }

    /// Reserve before browser allocation. Rejected values remain caller-owned.
    pub fn can_insert_pending(&self, bytes: usize) -> bool {
        bytes <= self.max_bytes
            && self.max_entries > 0
            && self
                .entries
                .values()
                .filter(|e| e.pending)
                .map(|e| e.bytes)
                .sum::<usize>()
                .saturating_add(bytes)
                <= self.max_bytes
            && self.entries.values().filter(|e| e.pending).count() < self.max_entries
    }

    pub fn insert_pending(&mut self, key: ImageKey, value: V, bytes: usize) -> bool {
        if self.entries.contains_key(&key) || !self.can_insert_pending(bytes) {
            return false;
        }
        self.make_room(bytes, 1);
        if !self.fits(bytes, 1) {
            return false;
        }
        self.put(key, Some(value), bytes, true);
        true
    }

    /// `is_value`가 가리키는 값을 가진 로드 중 항목의 키. 로드 이벤트가 어느 항목의 것인지 찾는다
    /// (내보냈거나 바뀐 값의 늦은 이벤트면 `None`).
    pub fn pending_key(&self, is_value: impl Fn(&V) -> bool) -> Option<ImageKey> {
        self.entries
            .iter()
            .find(|(_, e)| e.pending && e.value.as_ref().is_some_and(&is_value))
            .map(|(k, _)| *k)
    }

    /// Complete a reservation. Oversize resources transfer to a caller-owned ready lease;
    /// they never evict useful residents to obtain residency. Caller must release that lease.
    pub fn loaded(&mut self, key: &ImageKey, decoded_bytes: usize) -> Option<(V, usize)> {
        let entry = self.entries.get(key)?;
        if !entry.pending {
            return None;
        }
        let bytes = entry.bytes.saturating_add(decoded_bytes);
        let other_pending = self
            .entries
            .iter()
            .filter(|(k, e)| **k != *key && e.pending)
            .map(|(_, e)| e.bytes)
            .sum::<usize>();
        if bytes > self.max_entry_bytes || other_pending.saturating_add(bytes) > self.max_bytes {
            let mut entry = self.entries.remove(key)?;
            self.bytes -= entry.bytes;
            self.stats.oversize += 1;
            return entry.value.take().map(|v| (v, bytes));
        }
        let entry = self.entries.get_mut(key)?;
        entry.pending = false;
        self.bytes = self.bytes - entry.bytes + bytes;
        entry.bytes = bytes;
        self.make_room_except(0, 0, Some(*key));
        None
    }

    /// 로드 중이던 항목의 로드가 실패했다: 음성 항목으로 바꾼다(같은 바이트로 다시 로드하지 않음).
    pub fn failed(&mut self, key: &ImageKey) {
        if self.entries.get(key).is_some_and(|e| e.pending) {
            self.remove(key);
            self.insert(*key, None, NEGATIVE_ENTRY_BYTES);
        }
    }

    /// Reduce retained accounting to target_bytes, including cancellation of pending entries.
    /// A zero target releases every cache-owned reference; other Rc owners remain valid.
    pub fn trim(&mut self, target_bytes: usize) {
        while self.bytes > target_bytes || (target_bytes == 0 && !self.entries.is_empty()) {
            if !self.evict_any() {
                break;
            }
        }
    }

    /// 예산을 바꾼다. 줄이면 바로 맞춘다.
    pub fn set_budget(&mut self, max_bytes: usize, max_entries: usize) {
        self.max_bytes = max_bytes;
        self.max_entries = max_entries;
        self.max_entry_bytes = max_bytes / 4;
        let oversize: Vec<ImageKey> = self
            .entries
            .iter()
            .filter(|(_, e)| {
                !e.pending && !e.transient && e.value.is_some() && e.bytes > self.max_entry_bytes
            })
            .map(|(k, _)| *k)
            .collect();
        for key in oversize {
            self.evict(&key);
        }
        while !self.fits(0, 0) {
            if !self.evict_any() {
                break;
            }
        }
        self.frequency.retain(|k, _| self.entries.contains_key(k));
    }

    pub fn stats(&self) -> CacheStats {
        CacheStats {
            entries: self.entries.len(),
            negative_entries: self.entries.values().filter(|e| e.value.is_none()).count(),
            pending_entries: self.entries.values().filter(|e| e.pending).count(),
            transient_entries: self.entries.values().filter(|e| e.transient).count(),
            bytes: self.bytes,
            max_bytes: self.max_bytes,
            max_entries: self.max_entries,
            max_entry_bytes: self.max_entry_bytes,
            ..self.stats
        }
    }

    fn put(&mut self, key: ImageKey, value: Option<V>, bytes: usize, pending: bool) {
        self.tick += 1;
        self.bytes += bytes;
        self.entries.insert(
            key,
            Entry {
                value,
                bytes,
                used: self.tick,
                pending,
                transient: false,
            },
        );
    }

    /// 요청 횟수를 센다. 요청이 (보관 항목 수의 4배, 최소 64)번 쌓이면 모든 횟수를 반으로 줄인다. 셈표 크기는
    /// 항목 수 한도의 4배로 묶는다(넘으면 바로 줄이고, 그래도 넘으면 보관하지 않은 키의 횟수를 버린다).
    fn count_request(&mut self, key: &ImageKey) {
        if !self.admission {
            return;
        }
        self.since_aging += 1;
        if self.since_aging >= 4 * self.entries.len().max(16) {
            self.age();
        }
        let count = self.frequency.entry(*key).or_insert(0);
        *count = count.saturating_add(1).min(MAX_FREQUENCY);
        if self.frequency.len() > 4 * self.max_entries {
            self.age();
            if self.frequency.len() > 4 * self.max_entries {
                let entries = &self.entries;
                self.frequency.retain(|k, _| entries.contains_key(k));
            }
        }
    }

    fn age(&mut self) {
        self.since_aging = 0;
        self.frequency.retain(|_, count| {
            *count /= 2;
            *count > 0
        });
    }

    /// 입장 제한: `bytes`를 넣으려고 내보내야 할 항목(일시 항목·오래된 순) 모두보다 새 키의 요청이 많은가.
    /// 내보낼 것이 없으면(자리가 있음) 들인다.
    fn admits(&self, key: &ImageKey, bytes: usize) -> bool {
        let candidate = self.frequency.get(key).copied().unwrap_or(0);
        let mut victims: Vec<(&ImageKey, &Entry<V>)> =
            self.entries.iter().filter(|(_, e)| !e.pending).collect();
        victims.sort_by_key(|(_, e)| (!e.transient, e.used));
        let (mut total, mut count) = (self.bytes, self.entries.len());
        for (victim, entry) in victims {
            if total + bytes <= self.max_bytes && count < self.max_entries {
                return true;
            }
            if !entry.transient && self.frequency.get(victim).copied().unwrap_or(0) >= candidate {
                return false;
            }
            total -= entry.bytes;
            count -= 1;
        }
        true
    }

    fn fits(&self, bytes: usize, count: usize) -> bool {
        self.bytes.saturating_add(bytes) <= self.max_bytes
            && self.entries.len().saturating_add(count) <= self.max_entries
    }

    /// Count reclaimable resident ownership without cloning values or changing LRU.
    pub fn resident_capacity_where(&self, eligible: impl Fn(&V) -> bool) -> (usize, usize) {
        self.entries
            .values()
            .filter(|e| !e.pending && e.value.as_ref().is_some_and(&eligible))
            .fold((0, 0usize), |(count, bytes), e| {
                (count + 1, bytes.saturating_add(e.bytes))
            })
    }

    /// Remove one least-recently-used eligible resident. The predicate must not clone
    /// ownership while deciding eligibility; pending entries are never selected.
    pub fn evict_resident_where(&mut self, eligible: impl Fn(&V) -> bool) -> bool {
        let key = self
            .entries
            .iter()
            .filter(|(_, e)| !e.pending && e.value.as_ref().is_some_and(&eligible))
            .min_by_key(|(_, e)| e.used)
            .map(|(k, _)| *k);
        if let Some(key) = key {
            self.evict(&key);
            true
        } else {
            false
        }
    }

    fn evict_any(&mut self) -> bool {
        let key = self
            .entries
            .iter()
            .min_by_key(|(_, e)| (e.pending, e.used))
            .map(|(k, _)| *k);
        if let Some(key) = key {
            self.evict(&key);
            true
        } else {
            false
        }
    }

    fn make_room(&mut self, bytes: usize, count: usize) {
        self.make_room_except(bytes, count, None);
    }

    /// 로드 중 항목과 `keep`을 빼고, 일시 항목·오래된 것부터 내보내 `bytes`·`count`가 들어갈 자리를 만든다.
    fn make_room_except(&mut self, bytes: usize, count: usize, keep: Option<ImageKey>) {
        while !self.fits(bytes, count) {
            if !self.evict_victim(keep) {
                break;
            }
        }
    }

    fn evict_victim(&mut self, keep: Option<ImageKey>) -> bool {
        let Some(key) = self
            .entries
            .iter()
            .filter(|(k, e)| !e.pending && Some(**k) != keep)
            .min_by_key(|(_, e)| (!e.transient, e.used))
            .map(|(k, _)| *k)
        else {
            return false;
        };
        self.evict(&key);
        true
    }

    fn evict(&mut self, key: &ImageKey) {
        if self.remove(key) {
            self.stats.evictions += 1;
        }
    }

    fn remove(&mut self, key: &ImageKey) -> bool {
        let Some(entry) = self.entries.remove(key) else {
            return false;
        };
        self.bytes -= entry.bytes;
        if let Some(value) = entry.value {
            value.release();
        }
        true
    }
}

/// RGBA backing store 추정(오버플로는 포화).
pub fn rgba_bytes(width: u32, height: u32) -> usize {
    (width as usize)
        .saturating_mul(height as usize)
        .saturating_mul(4)
}

#[cfg(test)]
mod tests {
    use super::*;
    use std::cell::RefCell;
    use std::rc::Rc;

    #[derive(Clone)]
    struct Surface {
        id: u32,
        released: Rc<RefCell<Vec<u32>>>,
    }
    impl Release for Surface {
        fn release(&self) {
            self.released.borrow_mut().push(self.id);
        }
    }

    fn surface(id: u32, log: &Rc<RefCell<Vec<u32>>>) -> Surface {
        Surface {
            id,
            released: log.clone(),
        }
    }
    fn key(n: u8) -> ImageKey {
        ImageKey::of(&[n; 16])
    }

    #[test]
    fn byte_budget_evicts_least_recently_used_and_releases_it() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 16, false);
        assert!(cache.insert(key(1), Some(surface(1, &log)), 100));
        assert!(cache.insert(key(2), Some(surface(2, &log)), 100));
        assert!(cache.insert(key(3), Some(surface(3, &log)), 100));
        assert!(cache.get(&key(1)).is_some()); // 1을 최근으로
        assert!(cache.insert(key(4), Some(surface(4, &log)), 100));
        assert!(cache.insert(key(5), Some(surface(5, &log)), 100)); // 500 > 400: 가장 오래된 2를 내보냄
        assert_eq!(*log.borrow(), vec![2]);
        assert!(cache.get(&key(2)).is_none());
        assert!(cache.get(&key(1)).is_some());
        let stats = cache.stats();
        assert_eq!((stats.entries, stats.bytes, stats.evictions), (4, 400, 1));
    }

    #[test]
    fn oversize_entry_is_not_retained_and_does_not_evict_others() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 16, false);
        cache.insert(key(1), Some(surface(1, &log)), 100);
        assert!(!cache.insert(key(9), Some(surface(9, &log)), 101)); // 한도 400/4 = 100 초과
        assert!(
            log.borrow().is_empty(),
            "caller owns and releases the oversize value"
        );
        assert!(cache.get(&key(9)).is_none());
        assert_eq!(
            (
                cache.stats().entries,
                cache.stats().bytes,
                cache.stats().oversize
            ),
            (1, 100, 1)
        );
    }

    #[test]
    fn replacing_a_key_releases_the_previous_value_without_double_counting() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 16, false);
        cache.insert(key(1), Some(surface(1, &log)), 80);
        cache.insert(key(1), Some(surface(11, &log)), 60);
        assert_eq!(*log.borrow(), vec![1]);
        assert_eq!((cache.stats().entries, cache.stats().bytes), (1, 60));
        assert_eq!(cache.get(&key(1)).unwrap().unwrap().id, 11);
    }

    #[test]
    fn negative_entries_are_bounded_by_count_and_lru() {
        let log = Rc::new(RefCell::new(Vec::<u32>::new()));
        let mut cache: BoundedCache<Surface> = BoundedCache::new(1 << 20, 8, true);
        for n in 0..100u8 {
            cache.insert(key(n), None, 999_999); // 음성 항목 비용은 고정, 입장 제한 없음
        }
        let stats = cache.stats();
        assert_eq!(
            (stats.entries, stats.negative_entries, stats.bytes),
            (8, 8, 8 * NEGATIVE_ENTRY_BYTES)
        );
        assert!(matches!(cache.get(&key(99)), Some(None)));
        assert!(cache.get(&key(0)).is_none());
        assert!(log.borrow().is_empty());
    }

    #[test]
    fn trim_and_budget_reduction_release_everything_they_drop() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 16, false);
        for n in 1..=4 {
            cache.insert(key(n), Some(surface(n as u32, &log)), 100);
        }
        cache.insert(key(5), None, 0); // 400 + 64 > 400: 1을 내보냄 → 364
        assert_eq!(*log.borrow(), vec![1]);
        cache.trim(250); // 2, 3을 내보냄 → 4(100) + 음성(64)
        assert_eq!(
            (cache.stats().entries, cache.stats().bytes),
            (2, 100 + NEGATIVE_ENTRY_BYTES)
        );
        cache.set_budget(80, 16); // 항목 한도 20: 4는 이제 너무 큼. 음성 항목은 고정 비용이라 남는다
        assert_eq!(
            (cache.stats().entries, cache.stats().bytes),
            (1, NEGATIVE_ENTRY_BYTES)
        );
        cache.trim(0);
        let stats = cache.stats();
        assert_eq!((stats.entries, stats.bytes), (0, 0));
        let mut released = log.borrow().clone();
        released.sort();
        assert_eq!(released, vec![1, 2, 3, 4]);
    }

    #[test]
    fn content_key_distinguishes_length_and_bytes() {
        assert_ne!(ImageKey::of(b"abc"), ImageKey::of(b"abd"));
        assert_ne!(ImageKey::of(&[0u8; 3]), ImageKey::of(&[0u8; 4]));
        assert_eq!(ImageKey::of(b"same bytes"), ImageKey::of(b"same bytes"));
        assert_eq!(rgba_bytes(10, 20), 800);
        assert_eq!(rgba_bytes(u32::MAX, u32::MAX), usize::MAX);
    }

    // ---- 쓰는 중인 값·로드 중 항목 (교정: 원 패치는 그리기 직전 같은 이미지를 놓았다) ----

    #[test]
    fn ready_lease_is_transferred_without_resident_eviction() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut c = BoundedCache::new(400, 8, false);
        assert!(c.insert(key(1), Some(surface(1, &log)), 80));
        assert!(c.insert_pending(key(2), surface(2, &log), 10));
        let (lease, bytes) = c.loaded(&key(2), 800).unwrap();
        assert_eq!(bytes, 810);
        assert!(c.get(&key(1)).is_some());
        assert_eq!(c.stats().bytes, 80);
        assert!(log.borrow().is_empty());
        lease.release();
        assert_eq!(*log.borrow(), vec![2]);
    }

    #[test]
    fn pending_admission_rejects_and_trim_cancels() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut c = BoundedCache::new(100, 2, false);
        assert!(c.insert_pending(key(1), surface(1, &log), 90));
        assert!(!c.can_insert_pending(90));
        assert!(!c.insert_pending(key(2), surface(2, &log), 90));

        assert_eq!(*log.borrow(), vec![1]);

    }

    // ---- 입장 제한 (Stage B) ----

    fn scan(
        cache: &mut BoundedCache<Surface>,
        log: &Rc<RefCell<Vec<u32>>>,
        keys: std::ops::Range<u8>,
    ) -> usize {
        let mut hits = 0;
        for n in keys {
            if cache.get(&key(n)).is_some() {
                hits += 1;
            } else if !cache.insert(key(n), Some(surface(n as u32, log)), 100) {
                surface(n as u32, log).release(); // 호출자가 그린 뒤 놓는다
            }
        }
        hits
    }

    #[test]
    fn admission_keeps_a_stable_subset_when_a_scan_exceeds_the_budget() {
        // 예산 4개, 작업 집합 10개를 되풀이해 훑는다. LRU는 매번 모두 갈아 치워 적중 0, 입장 제한은 일부를 지킨다.
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut lru = BoundedCache::new(400, 64, false);
        let mut tiny = BoundedCache::new(400, 64, true);
        let (mut lru_hits, mut tiny_hits) = (0, 0);
        for _ in 0..5 {
            lru_hits += scan(&mut lru, &log, 0..10);
            tiny_hits += scan(&mut tiny, &log, 0..10);
        }
        assert_eq!(lru_hits, 0);
        assert!(tiny_hits >= 4 * 4, "tiny_hits={tiny_hits}");
        assert!(tiny.stats().bytes <= 400);
        assert!(tiny.stats().rejected > 0);
    }

    #[test]
    fn a_repeatedly_requested_newcomer_displaces_a_once_used_resident() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 64, true);
        scan(&mut cache, &log, 0..4); // 4개가 한 번씩 쓰여 가득 참
        assert_eq!(scan(&mut cache, &log, 9..10), 0); // 첫 요청: 같은 횟수(1)라 들이지 않음
        assert!(cache.get(&key(9)).is_none()); // 두 번째 요청(2 > 1)
        assert!(cache.insert(key(9), Some(surface(9, &log)), 100));
        assert!(cache.get(&key(9)).is_some());
    }

    #[test]
    fn aging_lets_a_new_working_set_replace_an_old_hot_one() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 64, true);
        for _ in 0..15 {
            scan(&mut cache, &log, 0..4); // 0..4가 아주 자주 쓰임
        }
        let mut admitted = false;
        for _ in 0..200 {
            scan(&mut cache, &log, 20..24); // 이제 다른 4개만 쓴다
            if (20..24).all(|n| cache.entries.contains_key(&key(n))) {
                admitted = true;
                break;
            }
        }
        assert!(admitted, "the new working set must eventually be admitted");
        assert!(cache.frequency.len() <= 4 * 64);
    }

    #[test]
    fn t010_burst_new_working_set_is_admitted_on_second_consecutive_request() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 64, true);
        for _ in 0..15 { scan(&mut cache, &log, 0..4); }
        assert!(cache.get(&key(9)).is_none());
        assert!(!cache.insert(key(9), Some(surface(9, &log)), 100));
        assert!(cache.get(&key(9)).is_none());
        assert!(cache.insert(key(9), Some(surface(9, &log)), 100),
            "a consecutive re-request must become resident instead of repeatedly decoding behind a historical hot set");
        assert!(cache.get(&key(9)).is_some());
        assert_eq!((cache.stats().entries, cache.stats().bytes), (4, 400));
    }

    #[test]
    fn t010_burst_interleaved_scan_does_not_override_hot_residents() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 64, true);
        for _ in 0..15 { scan(&mut cache, &log, 0..4); }
        for n in [9, 10, 9, 10] {
            assert!(cache.get(&key(n)).is_none());
            assert!(!cache.insert(key(n), Some(surface(n as u32, &log)), 100));
        }
        assert!((0..4).all(|n| cache.entries.contains_key(&key(n))));
        assert_eq!(cache.stats().bytes, 400);
    }

    #[test]
    fn t010_burst_never_bypasses_per_entry_or_zero_budget_limits() {
        let log = Rc::new(RefCell::new(Vec::new()));
        let mut cache = BoundedCache::new(400, 64, true);
        scan(&mut cache, &log, 0..4);
        for _ in 0..3 {
            assert!(cache.get(&key(9)).is_none());
            assert!(!cache.insert(key(9), Some(surface(9, &log)), 101));
        }
        assert_eq!((cache.stats().entries, cache.stats().bytes), (4, 400));
        cache.set_budget(0, 0);
        for _ in 0..3 {
            assert!(cache.get(&key(9)).is_none());
            assert!(!cache.insert(key(9), Some(surface(9, &log)), 100));
        }
        assert_eq!((cache.stats().entries, cache.stats().bytes), (0, 0));
    }
