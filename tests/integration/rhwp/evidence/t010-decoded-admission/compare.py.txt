from pathlib import Path
import json,hashlib
root=Path('<workspace>');out=root/'tests/integration/rhwp/evidence/t010-decoded-admission'

for name,d in [('baseline',before),('candidate',after)]:
 calls=[c for stage in d['rawStages'] for c in stage['calls'] if c['name']=='pollRenderJob']
 assert len({c['job'] for c in calls})==len(calls) and all(c['outcome']=='COMPLETE' for c in calls)

   assert c[side]['bytes']<=c[side]['maxBytes'] and c[side]['entries']<=c[side]['maxEntries']

 results.append({'arm':name,'runtimeIdentity':d['engineIdentity'],'sampleSha256':row['sourceSha256'],'perSampleWindowUniqueTerminalPolls':len(calls),'cacheDeltas':{k:sum(c['decodedAfter'][k]-c['decodedBefore'][k] for c in calls) for k in ['hits','misses','rejected','evictions']},'latenciesMs':{k:row[k] for k in ['groupStart','extend','undo','redo','afterUndo']},'rejectionRows':[{'job':c['job'],'page':c['page'],'pollMs':c['ms']} for c in calls if c['decodedAfter']['rejected']>c['decodedBefore']['rejected']]})
assert results[0]['sampleSha256']==results[1]['sampleSha256']
r={'status':'SCOPED_DIAGNOSTIC_COMPLETE','implementationChanges':changed,'sourceAndObserverControlled':True,'evidence':{n+'.json':hashlib.sha256((out/(n+'.json')).read_bytes()).hexdigest() for n in ['native-admission','candidate-admission']},'arms':results,'scope':'One chemical HWP trace per arm, unchanged canonical key/RAF protocol with extra observers.
