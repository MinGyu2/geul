"""Lifecycle cross-reference/gate-association analysis: new run + historical revalidation under the new contract."""
from pathlib import Path
import json,hashlib,sys,re,collections
root=Path('<workspace>/tests/integration/rhwp');ev=root/'evidence';out=ev/'t010-diagnostic-integrity'
sys.path.insert(0,str(root/'ownership-poc'));import lifecycle_evidence as L
W='48adaa0f748484d12f59f4805deda0480635f915368392a87e76e95962ec0b27'
def ref(p):return {'path':str(p.relative_to(root.parent.parent.parent)),'sha256':hashlib.sha256(p.read_bytes()).hexdigest()}
def canonical(d):return [(f"{doc['sample']}:{row['mode']}:{row['cycle']}:{l}",row[l]) for doc in d['measurements']['lifecycle'] for row in doc['rows'] for l in ('before','opened','closed','closedAfter10s') if l in row]
r={'newRun':{},'historical':{}}
new=out/'diagnostic-lifecycle-continuous-identity.json'
if new.exists():
  d=json.loads(new.read_text());v=L.classify(d,W);g=[x for x in d['gates'] if 'association' in x]
  r['newRun']={'source':ref(new),'status':v['status'],'failures':v['failures'],'unavailableEvidence':v['unavailableEvidence'],'runStatus':d.get('status'),'parentExit':d.get('parentExit'),
   'completedCycles':v['completedCycles'],'memoryPoints':v['memoryPoints'],'webProcessIdentity':v['webProcessIdentity'],
   'contentCrossCheck':{'durableSamples':len(d['samples']),'lifecycleObservations':len(canonical(d)),'identical':sum(1 for i,(w,p) in enumerate(canonical(d)) if i<len(d['samples']) and d['samples'][i]==p)},
   'gates':{'total':len(d['gates']),'associated':len(g),'cyclePass':sum(x['result']=='PASS' for x in d['gates']),'summaryInconclusive':sum(x['result']=='INCONCLUSIVE' for x in d['gates'])},
   'executedDependenciesStable':d.get('executedDependenciesStable'),'workspaceChangedDuringRun':d['workspaceMutationDuringRun']['changed'],'elapsedSeconds':d.get('elapsedSeconds'),
   'rssMiB':{'min':round(min(s['aggregateRssBytes'] for s in d['samples'])/2**20,1),'max':round(max(s['aggregateRssBytes'] for s in d['samples'])/2**20,1)},
   'cleanup':d.get('cleanup'),'engineIdentity':[d['engineIdentity']['wasmSha256'],d['engineIdentityAtEnd']['wasmSha256']]}
for name,path in (('6a58ff8-continuous',ev/'t010-native-completeness/diagnostic-lifecycle-continuous-identity.json'),('7efb162-continuous',ev/'t010-native-gate-followup/diagnostic-d4-lifecycle-continuous.json')):
  d=json.loads(path.read_text());v=L.classify(d,W)
  meta={'at','doc','sequence'}
  strip=lambda x:{k:v for k,v in x.items() if k not in meta} if isinstance(x,dict) else x
  c=canonical(d);same_without_meta=sum(1 for i,(w,p) in enumerate(c) if i<len(d['samples']) and strip(d['samples'][i])==strip(p))
  cats=collections.Counter()
  for f in v['failures']:
    if re.search(r'startTime|not bound to this document/session|memory observations missing|identity not stable|run cleanup incomplete|more observation failures|durable sample \d+ differs .*\[\'at\', \'doc\', \'sequence\'\]|more sample mismatches',f):cats['missing older-contract field']+=1
    else:cats['other: '+f[:80]]+=1
  r['historical'][name]={'source':ref(path),'newContractStatus':v['status'],'unavailableEvidence':v['unavailableEvidence'],'failureCategories':dict(cats),
   'contentEqualIgnoringRunnerMetadata':f'{same_without_meta}/{len(c)} (metadata keys at/doc/sequence excluded only for runs whose rows never stored them)',
   'gateOrderItemsResults':'checked by new classifier; association unavailable' }
r['interpretation']={'6a58ff8-continuous':'valid under its contract; under the new contract content and gate order/results all agree, only production-time gate association was never recorded -> INCONCLUSIVE, no regression',
  '7efb162-continuous':'valid under its contract; lacks start time, sample binding and gate association (cannot be reconstructed) -> new-contract FAIL caused only by unavailable fields; observation contents agree apart from runner metadata'}
(out/'lifecycle-cross-reference-analysis.json').write_text(json.dumps(r,ensure_ascii=False,indent=1)+'\n')
print(json.dumps({k:(v.get('status') if k=='newRun' else {n:(x['newContractStatus'],x['failureCategories'],x['contentEqualIgnoringRunnerMetadata'][:9]) for n,x in v.items()}) for k,v in r.items() if k!='interpretation'},ensure_ascii=False))
print(r['newRun'] and {k:r['newRun'][k] for k in ('status','failures','contentCrossCheck','gates','webProcessIdentity','executedDependenciesStable','workspaceChangedDuringRun','rssMiB')})
