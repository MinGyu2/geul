def diagnostic_install():
    script = native_probe.INSTALL
    start = script.index('  // Identical read-only Canvas timing')
    end = script.index('  p.selectionAt =', start)
    script = script[:start] + r'''
  const surfaceIds=new WeakMap();let surfaceSerial=0;
  const surface=c=>{
    if(!surfaceIds.has(c))surfaceIds.set(c,++surfaceSerial);
    return {id:surfaceIds.get(c),width:c.width,height:c.height,
      attached:!!c.isConnected,pageCanvas:!!c.classList?.contains('rhwp-page__canvas')};
  };
  for(const name of ['drawImage','getImageData','putImageData','clearRect']){
    const original=CanvasRenderingContext2D.prototype[name];
    CanvasRenderingContext2D.prototype[name]=function(...args){
      const at=performance.now(),phase=p.phase??null,target=surface(this.canvas);
      const source=name!=='drawImage'?null:args[0] instanceof HTMLCanvasElement
        ?{kind:'canvas',...surface(args[0])}:args[0] instanceof HTMLImageElement
        ?{kind:'image',width:args[0].naturalWidth,height:args[0].naturalHeight}:null;
      try{return original.apply(this,args);}
      finally{p.calls.push({name:'canvas:'+name,at,ms:performance.now()-at,phase,target,source});}
    };
  }
  const src=Object.getOwnPropertyDescriptor(HTMLImageElement.prototype,'src');
  if(src?.set){
    const imageStarts=new WeakMap();let imageSerial=0;
    Object.defineProperty(HTMLImageElement.prototype,'src',{...src,set(value){
      const id=++imageSerial,at=performance.now(),phase=p.phase??null;
      imageStarts.set(this,{id,at});
      const loaded=event=>{
        this.removeEventListener('load',loaded);this.removeEventListener('error',loaded);
        if(imageStarts.get(this)?.id!==id)return;
        p.calls.push({name:'image:'+event.type,at,ms:performance.now()-at,id,
          width:this.naturalWidth,height:this.naturalHeight,phase});
      };
      this.addEventListener('load',loaded);this.addEventListener('error',loaded);
      try{return src.set.call(this,value);}
      finally{p.calls.push({name:'image:src',at,ms:performance.now()-at,id,phase});}
    }});
  }
''' + script[end:]
    original = 'value=original.apply(this,args);completed=true;'
    assert script.count(original) == 1
    script = script.replace(original, 'const previous=p.phase;p.phase=name;try{value=original.apply(this,args);completed=true;}finally{p.phase=previous;}')
    return script.replace("import('/rhwp/rhwp.js')", 'Promise.resolve(window.__geulOwnership.probeEngine())').replace('/rhwp/rhwp_bg.wasm', '/__ownership_candidate__/rhwp_bg.wasm')


PNG_PARITY = r"""
  p.pngParity = async pages => {
    if(!active)throw new Error('No authoritative live document');
    const rows=[];
    for(const page of pages){
      mod.trimRendererCaches(0);
      const target=document.createElement('canvas'),reference=document.createElement('canvas'),keys=JSON.parse(active.getPageSourceImageKeys(page)).keys;
      if(keys.length!==1)throw new Error('Expected one public fixture image');
      const bytes=active.getSourceImageBytes(keys[0]),view=new DataView(bytes.buffer,bytes.byteOffset,bytes.byteLength);
      if(bytes[24]!==8||![0,2].includes(bytes[25]))throw new Error('Only opaque 8bit gray/RGB PNG experiment');
      const width=view.getUint32(16),height=view.getUint32(20),pieces=[bytes.slice(0,8)],removed=[];
      for(let at=8;at<bytes.length;){
        const length=view.getUint32(at),kind=String.fromCharCode(...bytes.slice(at+4,at+8)),end=at+12+length;
        if(end>bytes.length)throw new Error('Invalid PNG extent');
        if(kind==='tRNS')throw new Error('Transparency outside experiment scope');
        if(['gAMA','iCCP','sRGB','cHRM','cICP'].includes(kind))removed.push(kind);
        else pieces.push(bytes.slice(at,end));
        at=end;
      }
      try{
        active.renderPageToCanvas(page,target,0.5);
        const candidates=new Set(),table=wasm.__wbindgen_externrefs;
        for(let i=0;i<table.length;i++){
          const value=table.get(i),c=value instanceof HTMLCanvasElement?value:value instanceof CanvasRenderingContext2D?value.canvas:null;
          if(c&&c.width===width&&c.height===height)candidates.add(c);
        }
        if(candidates.size!==1)throw new Error('Expected one fresh Rust decoded source Canvas');
        reference.width=width;reference.height=height;reference.getContext('2d').drawImage([...candidates][0],0,0);
        const row={page,key:keys[0],width,height,removed,modes:[]};
        for(const [mode,parts] of [['original',[bytes]],['normalized',pieces]]){
          const url=URL.createObjectURL(new Blob(parts,{type:'image/png'})),image=new Image();
          const first=document.createElement('canvas'),second=document.createElement('canvas');
          try{
            const at=performance.now();image.src=url;await image.decode();const decodeMs=performance.now()-at;
            const comparisons=[];
            for(const [label,w,h] of [['natural',width,height],['display',414,585]]){
              first.width=second.width=w;first.height=second.height=h;
              first.getContext('2d').drawImage(reference,0,0,w,h);second.getContext('2d').drawImage(image,0,0,w,h);
              const a=first.getContext('2d').getImageData(0,0,w,h).data,b=second.getContext('2d').getImageData(0,0,w,h).data;
              let pixels=0,maxDelta=0;
              for(let i=0;i<a.length;i+=4){let delta=0;for(let k=0;k<4;k++)delta=Math.max(delta,Math.abs(a[i+k]-b[i+k]));if(delta){pixels++;maxDelta=Math.max(maxDelta,delta);}}
              comparisons.push({label,width:w,height:h,pixels,maxDelta});
            }
            row.modes.push({mode,decodeMs,comparisons});
          }finally{URL.revokeObjectURL(url);image.removeAttribute('src');first.width=first.height=second.width=second.height=0;}
        }
        rows.push(row);
      }finally{target.width=target.height=reference.width=reference.height=0;}
    }
    mod.trimRendererCaches(0);return rows;
  };
"""


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ('app', 'driver', 'webkit-driver', 'xvfb', 'xdotool', 'output'):
        parser.add_argument('--' + name, required=True, type=Path)
    args = parser.parse_args()
    if args.output.exists():
        raise RuntimeError('Preserve previous diagnostic output before repeating')
    result = {'status': 'RUNNING', 'scope': 'three source PNGs; original vs color-metadata-stripped browser decode against fresh Rust decoded Canvas; laboratory pixel test only',
              'protocol': 'source Canvas exact RGBA at natural/display sizes; no production renderer change or D4 acceptance claim', 'observerSourceSha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
    display = subprocess.Popen([str(args.xvfb), '-displayfd', '1', '-screen', '0', '1280x1024x24', '-nolisten', 'tcp'], stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, text=True)
    try:
        os.environ.update(DISPLAY=':' + display.stdout.readline().strip(), GDK_BACKEND='x11', WEBKIT_DISABLE_SANDBOX_THIS_IS_DANGEROUS='1', NO_AT_BRIDGE='1', GTK_A11Y='none', LIBGL_ALWAYS_SOFTWARE='1', WEBKIT_DISABLE_DMABUF_RENDERER='1')
        os.environ['PATH'] = str(args.webkit_driver.parent) + os.pathsep + os.environ['PATH']
        with tempfile.TemporaryDirectory(prefix='geul-render-phases-') as folder:
            work = Path(folder)
            for name in ('XDG_CONFIG_HOME', 'XDG_CACHE_HOME', 'TMPDIR'):
                target = work / name; target.mkdir(); os.environ[name] = str(target)
            fixtures = work / 'fixtures'
            subprocess.run(['node', str(ROOT / 'tests/fixtures/rhwp/generate.mjs'), str(fixtures)], check=True, capture_output=True)
            smoke.APP = str(args.app.resolve())
            smoke.DRIVER_LOG = '/tmp/geul-caret-png-parity-driver.log'
            with smoke.tauri_driver(str(args.driver.resolve()), 4489, str(work / 'data')) as start:
                driver = start()
                try:
                    smoke.wait_until(driver, "return !!document.querySelector('.document-bar__open')", timeout=30)
                    poc.open_document(driver, fixtures / 'basic-ko.hwpx')
                    native_probe.install(driver, diagnostic_install().replace('  p.selectionAt =', PNG_PARITY + '  p.selectionAt ='))
                    history.close_discard(driver, 'basic-ko.hwpx')
                    windows = subprocess.check_output([str(args.xdotool), 'search', '--onlyvisible', '--name', '^geul'], text=True).split()
                    subprocess.run([str(args.xdotool), 'windowfocus', '--sync', windows[-1]], check=True)
                    driver.native_key_tool = str(args.xdotool)
                    driver.busy_probe = False
                    fixture = next(p for p in history.LARGE if p.name == '연구노트(화학).hwp')
                    poc.open_document(driver, fixture);poc.set_zoom(driver,50);poc.click_paragraph_start(driver)
                    smoke.wait_until(driver,'return window.__geulOwnership.stats().sessions.every(s=>!s.active&&!s.queued)',timeout=30)
                    driver.call('POST','/timeouts',{'script':180000})
                    result['pngParity'] = driver.call('POST','/execute/async',{'script':"const done=arguments[arguments.length-1];window.__rhwpProbe.pngParity([0,5,23]).then(done,e=>done({error:String(e)}));",'args':[]})
                    if not isinstance(result['pngParity'],list):raise RuntimeError(str(result['pngParity']))
                    history.close_discard(driver,fixture.name)
                    result['engineIdentity'] = driver.js('return window.__geulOwnership.identity')
                    result['adapterCounters'] = driver.js('return window.__geulOwnership.counters')
                    result['sourceSha256'] = hashlib.sha256(fixture.read_bytes()).hexdigest()
                finally:
                    smoke.end_session(driver)
        result['measurementComplete'] = len(result['pngParity'])==3
        result['status'] = 'PASS' if result['measurementComplete'] else 'FAIL'
    except Exception as error:
        result.update(status='FAIL', error=str(error))
    finally:
        display.terminate(); display.wait(timeout=10)
        result.update(measurements=history.REPORT, gates=history.GATES)
        result['officialGateStatus'] = 'NOT_EXECUTED'
        clean = json.dumps(result, ensure_ascii=False, indent=2).replace(str(ROOT), '<workspace>').replace(str(Path.home()), '<home>')
        args.output.write_text(clean + '\n')
    return 0 if result['status'] == 'PASS' else 1


if __name__ == '__main__':
    raise SystemExit(main())
