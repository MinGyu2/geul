#!/usr/bin/env python3
"""Diagnostic Canvas attribution for one native D4 fixture; no acceptance claim.

Uses the canonical scenario and frame boundary with additional observer overhead.
Records operation names, dimensions, IDs and timings, never image/document data.
The existing probe, scenario, candidate binary and product files stay unchanged.
"""
import argparse
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile
import shutil
import time

HERE = Path('<workspace>/tests/integration/rhwp/ownership-poc')
ROOT = HERE.parents[3]
sys.path.insert(0, str(HERE))
import native
import native_probe
import undo_native as history
from native import smoke, poc


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    for name in ('app', 'driver', 'webkit-driver', 'xvfb', 'xdotool', 'output'):
        parser.add_argument('--' + name, required=True, type=Path)
    args = parser.parse_args()
    if args.output.exists():
        raise RuntimeError('Preserve previous diagnostic output before repeating')
    result = {'status': 'RUNNING', 'scope': 'independent HWPX UndoRedo20-cycle window with direct native transport after tauri session creation; no concatenation or RSS acceptance claim',
              'protocol': 'same lifecycle UndoRedo per-cycle operations/sleeps/memory boundaries; only one mode; durable per-cycle saves', 'observerSourceSha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
    def save():
        clean=json.dumps(result,ensure_ascii=False,indent=2).replace(str(ROOT),'<workspace>').replace(str(Path.home()),'<home>')
        temporary=args.output.with_name(args.output.name+'.update.'+str(os.getpid()))
        temporary.write_text(clean+'\n');os.replace(temporary,args.output)
    display = subprocess.Popen([str(args.xvfb), '-displayfd', '1', '-screen', '0', '1280x1024x24', '-nolisten', 'tcp'], stdout=subprocess.PIPE, stderr=subprocess.DEVNULL, text=True)
    try:
        os.environ.update(DISPLAY=':' + display.stdout.readline().strip(), GDK_BACKEND='x11', WEBKIT_DISABLE_SANDBOX_THIS_IS_DANGEROUS='1', NO_AT_BRIDGE='1', GTK_A11Y='none', LIBGL_ALWAYS_SOFTWARE='1', WEBKIT_DISABLE_DMABUF_RENDERER='1')
        os.environ['PATH'] = str(args.webkit_driver.parent) + os.pathsep + os.environ['PATH']
        with tempfile.TemporaryDirectory(prefix='geul-render-phases-') as folder:
            work = Path(folder)
            for name in ('XDG_CONFIG_HOME', 'XDG_CACHE_HOME', 'TMPDIR'):
                target = work / name; target.mkdir(); os.environ[name] = str(target)
            fixtures = work / 'fixtures'
            subprocess.run(['node', str(ROOT / 'tests/fixtures/rhwp/generate.mjs'), str(fixtures)], check=True, capture_output=True)
            smoke.APP = str(args.app.resolve())
            smoke.DRIVER_LOG = '/tmp/geul-caret-lifecycle-direct-driver.log'
            with smoke.tauri_driver(str(args.driver.resolve()), 4489, str(work / 'data')) as start:
                driver = start()
                driver.base = 'http://127.0.0.1:4445'
                result['transport'] = 'tauri session creation then direct native WebKitWebDriver4445, same session ID; no request retries'
                try:
                    smoke.wait_until(driver, "return !!document.querySelector('.document-bar__open')", timeout=30)
                    poc.open_document(driver, fixtures / 'basic-ko.hwpx')
                    native_probe.install(driver, native_probe.INSTALL.replace("import('/rhwp/rhwp.js')", 'Promise.resolve(window.__geulOwnership.probeEngine())').replace('/rhwp/rhwp_bg.wasm', '/__ownership_candidate__/rhwp_bg.wasm'))
                    history.close_discard(driver, 'basic-ko.hwpx')
                    windows = subprocess.check_output([str(args.xdotool), 'search', '--onlyvisible', '--name', '^geul'], text=True).split()
                    subprocess.run([str(args.xdotool), 'windowfocus', '--sync', windows[-1]], check=True)
                    driver.native_key_tool = str(args.xdotool)
                    driver.busy_probe = False
                    result['engineIdentity'] = driver.js('return window.__geulOwnership.identity')
                    result['rows'] = []
                    result['cleanupChecks'] = []
                    fixture = next(p for p in history.LARGE if p.suffix=='.hwpx')
                    copy=work/fixture.name;shutil.copyfile(fixture,copy)
                    result['sourceSha256']=hashlib.sha256(fixture.read_bytes()).hexdigest()
                    def memory():
                        return {**history.memory_sample(driver),'renderer':driver.js('return window.__geulOwnership.stats()')}
                    driver.js('window.__rhwpProbe.historyDisabled=false')
                    save()
                    poc.set_zoom(driver,50)
                    for cycle in range(20):
                        before=memory();poc.open_document(driver,copy,expected='ready')
                        smoke.wait_until(driver,"return document.querySelector('.rhwp-page__canvas')?.width>0",timeout=30)
                        poc.click_paragraph_start(driver)
                        for _ in range(6):poc.press(driver,'Right');poc.type_ascii(driver,'a')
                        for _ in range(6):history.history_key(driver,'z');history.history_key(driver,'y')
                        time.sleep(1);opened=memory();history.close_discard(driver,copy.name);time.sleep(2);closed=memory()
                        row={'mode':'undoRedo','cycle':cycle,'before':before,'opened':opened,'closed':closed}
                        if cycle==19:time.sleep(8);row['closedAfter10s']=memory()
                        result['rows'].append(row)
                        result['cleanupChecks'].append({'cycle':cycle,'status':'PASS' if not closed['owners'] and closed['canvasCount']==0 and not closed['renderer']['sessions'] and closed['renderer']['subscribers']==0 else 'FAIL'})
                        save();print('hwpxUndoRedo',cycle,result['cleanupChecks'][-1]['status'],flush=True)
                finally:
                    smoke.end_session(driver)
        result['measurementComplete']=len(result['rows'])==20
        result['status']='PASS' if result['measurementComplete'] and all(r['status']=='PASS' for r in result['cleanupChecks']) else 'FAIL'
    except Exception as error:
        result.update(status='FAIL', error=str(error))
        if callable(getattr(error,'diagnostic',None)):result['driverDiagnostic']=error.diagnostic()
    finally:
        display.terminate(); display.wait(timeout=10)
        result.update(measurements=history.REPORT, gates=history.GATES)
        result['officialGateStatus'] = 'INCONCLUSIVE'
        clean = json.dumps(result, ensure_ascii=False, indent=2).replace(str(ROOT), '<workspace>').replace(str(Path.home()), '<home>')
        args.output.write_text(clean + '\n')
    return 0 if result['status'] == 'PASS' else 1


if __name__ == '__main__':
    raise SystemExit(main())
