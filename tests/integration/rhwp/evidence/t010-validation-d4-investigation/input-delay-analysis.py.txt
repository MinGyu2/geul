"""Attribute chemical HWP key->input delay with busy-probe gaps, absolute key/input times and probe/trace logs."""
import json,hashlib,sys,collections
from pathlib import Path
p=Path(sys.argv[1]);r=json.loads(p.read_text())
gs=[s for s in r['samples'] if s['kind']=='groupStart']
keys=[e['t'] for e in r['events']['keys']];inputs=[e['t'] for e in r['events']['inputs']]
calls=sorted(r['probeCalls'],key=lambda c:c['at'] or 0);spans=[s for s in r['traceSpans'] if isinstance(s.get('at'),(int,float))]
def pct(v):
  v=sorted(v);q=lambda x:v[min(len(v)-1,int(round(x*(len(v)-1))))]
  return {'n':len(v),'p50':round(q(.5),1),'p95':round(q(.95),1),'max':round(v[-1],1)} if v else None
rows=[]
for s in gs:
  a=s['at'];d=a['input'];busy=s.get('busy') or []
  # busy gaps (relative to key) overlapping the key->input window
  over=[g for g in busy if g[0]<d and g[0]+g[1]>0]
  rows.append({'keyToFrame':s['keyToFrame'],'delivery':d,'busyOverlapMs':round(sum(min(g[0]+g[1],d)-max(g[0],0) for g in over),1),'busyGaps':over[:4]})
slow=[x for x in rows if x['delivery']>=50]
# Absolute view: for each keydown with a matching input >=50ms later, list calls/spans that ran inside [key, input].
pairs=[];j=0
for k in keys:
  while j<len(inputs) and inputs[j]<k:j+=1
  if j<len(inputs):pairs.append((k,inputs[j]))
inside=collections.Counter();inside_ms=collections.Counter();examples=[]
for k,i in pairs:
  if i-k<50:continue
  cs=[c for c in calls if c['at'] is not None and k<=c['at']<i];ss=[x for x in spans if k<=x['at']<i]
  for c in cs:inside[c['name']]+=1;inside_ms[c['name']]+=c['ms'] or 0
  for x in ss:inside['span:'+x['name']]+=1;inside_ms['span:'+x['name']]+=x['ms'] or 0
  if len(examples)<4:examples.append({'keyToInput':round(i-k,1),'calls':[{'name':c['name'],'at':round(c['at']-k,1),'ms':round(c['ms'],1),'outcome':c.get('outcome')} for c in cs][:12],'spans':[{'name':x['name'],'at':round(x['at']-k,1),'ms':x['ms']} for x in ss][:12]})
out={'source':{'path':'tests/integration/rhwp/evidence/t010-validation-d4-investigation/'+p.name,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()},
 'status':r['status'],'official':False,'measureGroupStart':r['measure']['groupStart'],
 'delivery':pct([x['delivery'] for x in rows]),'slowSamples':len(slow),'slowWithBusyOverlap':sum(x['busyOverlapMs']>=0.8*x['delivery'] for x in slow),
 'slowRows':slow[:10],'absoluteSlowPairs':sum(1 for k,i in pairs if i-k>=50),'workInsideSlowKeyToInput':{n:{'count':inside[n],'ms':round(inside_ms[n],1)} for n,_ in inside.most_common(12)},'examples':examples,
 'traceDropped':r.get('traceDropped')}
Path(sys.argv[2]).write_text(json.dumps(out,ensure_ascii=False,indent=1)+'\n')
print(json.dumps({k:out[k] for k in ('status','measureGroupStart','delivery','slowSamples','slowWithBusyOverlap','absoluteSlowPairs','workInsideSlowKeyToInput')},ensure_ascii=False))
for e in out['examples'][:3]:print(json.dumps(e,ensure_ascii=False)[:900])
