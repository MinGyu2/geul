"""Decompose official D4 group-start samples (existing 48adaa official run) into key->revision components."""
import json,hashlib,sys
from pathlib import Path
ev=Path('<workspace>/tests/integration/rhwp/evidence');src=ev/'t010-native-gate-followup/d4-official-48adaa-full.json'
r=json.loads(src.read_text())
def pct(v):
  v=sorted(x for x in v if x is not None)
  if not v:return None
  q=lambda p:v[min(len(v)-1,int(round(p*(len(v)-1))))]
  return {'n':len(v),'p50':round(q(.5),1),'p95':round(q(.95),1),'max':round(v[-1],1)}
def ms(s,name):return (s['calls'].get(name) or {}).get('ms',0.0)
out={'source':{'path':str(src.relative_to(ev.parent.parent.parent.parent)),'sha256':hashlib.sha256(src.read_bytes()).hexdigest()},
 'method':'existing official MEASURE timeline + read-only probe call timings (performance.now on the WebView main thread; ms rounded); no new run',
 'definitions':{'delivery':'native keydown event timeStamp -> input event','handler':'input handler start -> end (synchronous)',
  'insertText':'engine edit call incl. its internal relayout','snapshot':'saveSnapshot+snapshotCapacity','pageInfo':'getPageInfo calls (count = pages)',
  'render':'pollRenderJob (synchronous render job inside the handler)','canvas':'visible draw + readback','otherHandler':'handler - instrumented engine/canvas calls inside it (app/React/JS)',
  'postHandlerToRevisionPoll':'handler end -> RAF poll that saw the revision','postRevision':'poll RAF -> end (two RAFs)'},'documents':[]}
for row in r['measurements']['perf']:
  gs=[s for s in row['timelineSamples'] if s['kind']=='groupStart']
  comp={k:[] for k in ('keyToFrame','delivery','handler','insertText','snapshot','pageInfo','pageInfoCalls','render','canvas','otherHandler','postHandlerToRevisionPoll','postRevision')}
  for s in gs:
    a=s['at'];h=(a['handlerEnd']-a['input']) if a['handlerEnd'] is not None else None
    inst=ms(s,'insertText')+ms(s,'saveSnapshot')+ms(s,'snapshotCapacity')+ms(s,'getPageInfo')+ms(s,'pollRenderJob')+ms(s,'canvasVisibleDraw')+ms(s,'canvasReadback')
    comp['keyToFrame'].append(a['end']);comp['delivery'].append(a['input']);comp['handler'].append(h)
    comp['insertText'].append(ms(s,'insertText'));comp['snapshot'].append(ms(s,'saveSnapshot')+ms(s,'snapshotCapacity'))
    comp['pageInfo'].append(ms(s,'getPageInfo'));comp['pageInfoCalls'].append((s['calls'].get('getPageInfo') or {}).get('n',0))
    comp['render'].append(ms(s,'pollRenderJob'));comp['canvas'].append(ms(s,'canvasVisibleDraw')+ms(s,'canvasReadback'))
    comp['otherHandler'].append(None if h is None else h-inst)
    comp['postHandlerToRevisionPoll'].append(None if a['handlerEnd'] is None else a['observed']-a['handlerEnd']);comp['postRevision'].append(a['end']-a['observed'])
  # tail attribution: the 3 slowest samples' dominant component
  slow=sorted(range(len(gs)),key=lambda i:-comp['keyToFrame'][i])[:3]
  tails=[{k:comp[k][i] for k in comp} for i in slow]
  out['documents'].append({'sample':row['sample'],'pages':row['pages'],**{k:pct(v) for k,v in comp.items()},'slowest3':tails})
(Path(sys.argv[1])).write_text(json.dumps(out,ensure_ascii=False,indent=1)+'\n')
for d in out['documents']:
  print(d['sample'][:12],d['pages'],'K2F',d['keyToFrame']['p50'],d['keyToFrame']['p95'],'deliv',d['delivery']['p50'],d['delivery']['p95'],'handler',d['handler']['p50'],d['handler']['p95'],'ins',d['insertText']['p50'],d['insertText']['p95'],'pInfo',d['pageInfo']['p50'],d['pageInfoCalls']['p50'],'render',d['render']['p50'],d['render']['p95'],'canvas',d['canvas']['p50'],'other',d['otherHandler']['p50'],d['otherHandler']['p95'],'post2raf',d['postRevision']['p50'],d['postRevision']['p95'])
  for t in d['slowest3']:print('   slow',{k:t[k] for k in ('keyToFrame','delivery','handler','insertText','render','otherHandler','postRevision')})
