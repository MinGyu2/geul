"""Mutation check for performance-visual-native.py acceptance rules (run from verified workspace root)."""
import sys,shutil,subprocess,json,pathlib,re
src=pathlib.Path('tests/integration/rhwp/ownership-poc').resolve();d=src.parent/'ownership-poc-mutation-tmp'
assert not d.exists()
muts={
 'baseline-unmutated':None,
 'documents-not-required':("    if documents != selected: failures.append","    if False: failures.append"),
 'references-not-required':("    if names != expected: failures.append","    if False: failures.append"),
 'unverified-allowed':("    if unverified: failures.append","    if False: failures.append"),
 'pixel-mismatch-allowed':("    if mismatched: failures.append","    if False: failures.append"),
 'cleanup-not-required':("    if incomplete: failures.append","    if False: failures.append"),
 'history-gates-ignored':("    if gates: failures.append","    if False: failures.append"),
 'unknown-sample-accepted':("    if sample not in FIXTURES: raise","    if False: raise"),
 'empty-selection-allowed':("    if not selected: failures.append","    if False: failures.append"),
 'failed-assertions-ignored':("    if failed: failures.append","    if False: failures.append"),
 'missing-assertions-ignored':("    if missing: failures.append","    if False: failures.append"),
}
out={}
try:
  for name,m in muts.items():
    shutil.copytree(src,d,ignore=shutil.ignore_patterns('__pycache__'))
    f=d/'performance-visual-native.py';t=f.read_text()
    if m: assert t.count(m[0])==1,name;f.write_text(t.replace(m[0],m[1]))
    p=subprocess.run([sys.executable,'-I','-m','unittest','discover','-s',str(d),'-p','test_performance_visual_native.py'],capture_output=True,text=True)
    out[name]={'exit':p.returncode,'summary':[l for l in p.stderr.splitlines() if l.startswith(('OK','FAILED'))][-1],'killedBy':re.findall(r'^(?:FAIL|ERROR): (\w+)',p.stderr,re.M)}
    shutil.rmtree(d)
finally:
  if d.exists(): shutil.rmtree(d)
survivors=[k for k,v in out.items() if k!='baseline-unmutated' and v['exit']==0]
print(json.dumps({'status':'PASS' if out['baseline-unmutated']['exit']==0 and not survivors else 'FAIL','survivors':survivors,'mutants':out},indent=1))
