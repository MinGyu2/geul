def pct(v,p):
    v=sorted(x for x in v if x is not None);return v[max(0,math.ceil(p/100*len(v))-1)] if v else None
def stat(v):
    v=[x for x in v if x is not None];return {'n':len(v),'p50':pct(v,50),'p95':pct(v,95),'max':max(v) if v else None}
def d(a,b): return None if a is None or b is None else a-b
def stages(path):
    data=json.loads(path.read_text());out=[]
    for i,x in enumerate(data['rawStages']):
        a=x['at'];calls=x['calls'];polls=[c for c in calls if c.get('name')=='pollRenderJob']
        miss=sum(c['decodedAfter']['misses']-c['decodedBefore']['misses'] for c in polls if 'decodedAfter' in c) if any('decodedAfter' in c for c in polls) else None
        upload=sum(c.get('name')=='canvas:putImageData' for c in calls)
        lo,hi=a.get('input'),a.get('handlerEnd')
        inside=[c for c in calls if lo is not None and hi is not None and lo<=c['at']<=hi and not str(c.get('name')).startswith('canvas:') and c.get('name')!='adapterRenderRequest']
        instrumented=sum(c.get('ms') or 0 for c in inside)
        out.append({'index':i,'kind':x['kind'],'keyToFrameMs':x['keyToFrame'],'handlerInstrumentedWasmMs':instrumented if hi is not None else None,
          'handlerUninstrumentedMs':(hi-lo-instrumented) if hi is not None and lo is not None else None,
          'nativeKeyToInputMs':d(a.get('input'),a.get('key')),'inputHandlerMs':d(a.get('handlerEnd'),a.get('input')),'handlerToRevisionObservedMs':d(a.get('observed'),a.get('handlerEnd')),
          'revisionToRaf1Ms':d(a.get('raf1'),a.get('observed')),'raf1ToRaf2EndMs':d(a.get('end'),a.get('raf1')),
          'pollWallMs':sum(c['ms'] for c in polls),'layerTreeWallMs':sum(c['ms'] for c in calls if c.get('name')=='getPageLayerTreeWithProfile'),
          'pageInfoWallMs':sum(c['ms'] for c in calls if c.get('name')=='getPageInfo'),'canvasCallWallMs':sum(c['ms'] for c in calls if str(c.get('name')).startswith('canvas:')),
          'imageUploads':upload,'decodedMisses':miss,'instrumentedWorkBeforeInputMs':sum((c.get('ms') or 0) for c in calls if a.get('input') is not None and c['at']<a['input']),'missingMarks':sorted(k for k,v in a.items() if v is None)})
    return data,out
def summarize(rows,kind):
    g=[r for r in rows if r['kind']==kind];keys=['keyToFrameMs','nativeKeyToInputMs','inputHandlerMs','handlerInstrumentedWasmMs','handlerUninstrumentedMs','handlerToRevisionObservedMs','revisionToRaf1Ms','raf1ToRaf2EndMs','pollWallMs','layerTreeWallMs','pageInfoWallMs','canvasCallWallMs']
    s={k:stat([r[k] for r in g]) for k in keys}
    up=[r for r in g if r['imageUploads']];noup=[r for r in g if not r['imageUploads']]
    s['samplesWithImageUpload']={'n':len(up),'keyToFrameMs':stat([r['keyToFrameMs'] for r in up]),'pollWallMs':stat([r['pollWallMs'] for r in up])}
    s['samplesWithoutImageUpload']={'n':len(noup),'keyToFrameMs':stat([r['keyToFrameMs'] for r in noup]),'nativeKeyToInputMs':stat([r['nativeKeyToInputMs'] for r in noup])}
    s['samplesOver50ms']=len([r for r in g if r['keyToFrameMs']>50])
    s['samplesWithMissingIntervalMarks']=sum(bool(r['missingMarks']) for r in g)
    return s
def decode_split(path):
    st=json.loads(path.read_text())['rawStages'];seen=collections.Counter();rows=[]
    for si,x in enumerate(st):
        for c in x['calls']:
            if c.get('name')!='pollRenderJob' or 'decodedAfter' not in c:continue
            b,a=c['decodedBefore'],c['decodedAfter'];miss=a['misses']-b['misses']
            rows.append({'kind':x['kind'],'page':c['page'],'ms':c['ms'],'miss':miss,'rejected':a['rejected']-b['rejected'],'repeatPage':miss>0 and seen[c['page']]>0})
            if miss:seen[c['page']]+=1
    m=[r for r in rows if r['miss']]
    return {'polls':len(rows),'missPolls':len(m),'firstMissForPageInWindow':stat([r['ms'] for r in m if not r['repeatPage']]),'repeatedMissForPage':stat([r['ms'] for r in m if r['repeatPage']]),
            'noMissPolls':stat([r['ms'] for r in rows if not r['miss']]),'missKinds':dict(collections.Counter(r['kind'] for r in m)),'repeatedPages':dict(collections.Counter(str(r['page']) for r in m if r['repeatPage'])),
            'repeatedDecodePollWallMsTotal':sum(r['ms'] for r in m if r['repeatPage'])}
def counterfactual(rows):
    g=[r for r in rows if r['kind']=='groupStart'];base=stat([r['keyToFrameMs'] for r in g])

    return {'groupStartObserved':base,'withoutUploadPollExcessOver13ms':stat(noUploadExcess),
